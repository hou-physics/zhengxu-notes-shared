step2_imaging.py (Scenario A only)

Combined-cube tclean imaging in CASA: produces one cube covering both CO(2-1) and H30α so that the strong-tracer mask aligns pixel-for-pixel with the HRL data. Only used when re-imaging from MS is justified (Scenario A); for the 32 sources in the final sample, the pipeline product was used directly and this step was skipped.

Key parameters (matched to literature pipelines): weighting='natural' for maximum sensitivity, width='10 km/s', restfreq='230.538 GHz' (CO reference frame), niter=10000, threshold='3 mJy', usemask='auto-multithresh', restoringbeam='common' so the cube has a single uniform beam.

##############################################################################
# Step 2: Combined Cube Imaging (CO + H30α)
#
# Usage: run inside CASA 6.6
#   cd /Volumes/HOU/NGC_3628/ALMA_data/2013.1.00087.S/.../manual_calibration
#   casa --nologger --nogui
#   execfile('/path/to/step2_imaging.py')
#
# Strategy: produce a single combined cube containing both CO(2-1) and
# H30α
#   - avoids grid-mismatch problems between separate cubes
#   - the CO mask transfers to H30α with pixel-exact alignment
#   - velocity range: -1500 to +1500 km/s (CO rest frequency as reference)
#
# References:
#   - the previous tclean parameter set (context document)
#   - Ada's parameters in proc-commands.py
#   - David Lau, thesis Section 2.2.1
##############################################################################

import os
import time

# ============================================================================
# Configuration
# ============================================================================

vis = 'NGC3628_concatenated.ms.contsub'
# tclean output goes to local SSD (avoids macOS ._sidecar issues on the
# external drive)
local_workdir = os.path.expanduser('~/tclean_tmp/NGC3628')
final_workdir = '/Volumes/HouAstro/master/master_thesis/work_dir/NGC3628'
workdir = final_workdir  # plots and logs still live on the external drive

if not os.path.exists(local_workdir):
    os.makedirs(local_workdir)

imagename = os.path.join(local_workdir, 'NGC3628_CO_H30alpha_v5')

# Sanity-check input
if not os.path.exists(vis):
    print("ERROR: {} not found!".format(vis))
    print("Run step1_uvcontsub.py first.")
    import sys; sys.exit(1)

# Make sure the working directory exists
if not os.path.exists(workdir):
    os.makedirs(workdir)

print("="*70)
print("  Step 2: Combined Cube Imaging")
print("="*70)
print("")
print("Input:  {}".format(vis))
print("Output: {}".format(imagename))
print("")

# ============================================================================
# Clean up any previous output
# ============================================================================
for ext in ['.image', '.model', '.residual', '.pb', '.psf', '.sumwt',
            '.mask', '.fits']:
    old = imagename + ext
    if os.path.exists(old):
        print("removing previous {}".format(os.path.basename(old)))
        os.system('rm -rf ' + old)

# ============================================================================
# tclean parameter notes
# ============================================================================
#
# Velocity range design:
#   NGC 3628 systemic velocity ~830 km/s (z = 0.002772)
#   CO rest frequency  = 230.538 GHz
#   H30α rest frequency = 231.901 GHz
#
#   In the CO rest frame,
#   v(H30α) ≈ c × (230.538 - 231.901 × (1 - 0.002772)) / 230.538
#          ≈ c × (230.538 - 231.258) / 230.538
#          ≈ -937 km/s (relative to CO)
#
#   So the cube needs to cover:
#   - CO   around +700 .. +1000 km/s  (systemic velocity region)
#   - H30α around  -900 ..  -600 km/s
#
#   The range -1500 to +1500 km/s comfortably covers both lines.
#
# Imaging parameters:
#   - specmode='cube':          spectral cube
#   - width='10km/s':           10 km/s per channel (matches Ada)
#   - nchan=300:                covers 3000 km/s
#   - restfreq='230.538GHz':    CO rest frequency as reference
#   - outframe='LSRK':          standard reference frame
#   - weighting='natural':      max sensitivity / recover extended
#                               emission (matches Ada / David)
#   - niter=10000, threshold='3mJy':  Ada / David working values
#   - deconvolver='hogbom':     standard point-source deconvolver
#   - usemask='auto-multithresh':  CASA adaptive automask (close to
#                                  interactive in practice)
#   - pbcor=True:               emit the pbcor cube directly
#
# ============================================================================

print("starting tclean ...")
print("(1-3 hours depending on machine)")
print("")

t_start = time.time()

tclean(
    vis         = vis,
    imagename   = imagename,
    field       = 'NGC_3628',
    spw         = '0,1,4,5',         # SPWs containing CO and H30α
    specmode    = 'cube',
    start       = '-1500km/s',
    width       = '10km/s',
    nchan       = 300,
    restfreq    = '230.538GHz',       # CO(2-1) rest frequency
    outframe    = 'LSRK',
    veltype     = 'optical',
    imsize      = [512, 512],
    cell        = '0.2arcsec',
    weighting   = 'natural',           # matches Ada / David, max sensitivity
    deconvolver = 'hogbom',
    niter       = 10000,              # matches Ada / David (was 100000, over-cleaning)
    threshold   = '3mJy',             # avoid over-cleaning
    usemask     = 'auto-multithresh', # adaptive automask, replaces interactive
    interactive = False,
    pbcor       = True,               # emit pbcor cube directly
    restoringbeam = 'common'          # single beam across the cube, eases analysis
)

t_end = time.time()
elapsed = (t_end - t_start) / 60.0

print("")
print("tclean finished in {:.1f} min".format(elapsed))
print("")

# ============================================================================
# Export FITS
# ============================================================================

if os.path.exists(imagename + '.image'):
    print("exporting FITS files ...")
    # In CASA 6.x with tclean(pbcor=True) the outputs are:
    #   .image       -> always non-pbcor (raw Jy/beam)
    #   .image.pbcor -> the true pbcor cube
    #   .pb          -> primary-beam response
    # So the pbcor FITS must be exported from .image.pbcor!

    # Export the pbcor cube (used for flux integration)
    fitsname = imagename + '_pbcor.fits'
    exportfits(
        imagename = imagename + '.image.pbcor',
        fitsimage = fitsname,
        overwrite = True
    )
    print("FITS (pbcor):     {}".format(fitsname))

    # Also export the non-pbcor cube (used to build the mask)
    fitsname_nonpbcor = imagename + '_nonpbcor.fits'
    exportfits(
        imagename = imagename + '.image',
        fitsimage = fitsname_nonpbcor,
        overwrite = True
    )
    print("FITS (non-pbcor): {}".format(fitsname_nonpbcor))

    # Export the PB response (used to generate the non-pbcor mask)
    if os.path.exists(imagename + '.pb'):
        pb_fitsname = imagename + '_pb.fits'
        exportfits(
            imagename = imagename + '.pb',
            fitsimage = pb_fitsname,
            overwrite = True
        )
        print("PB FITS:          {}".format(pb_fitsname))
    else:
        print("WARNING: .pb image not found!")

    # Copy FITS to the external drive
    import shutil
    for src in [fitsname, fitsname_nonpbcor, pb_fitsname]:
        if os.path.exists(src):
            dst = os.path.join(final_workdir, os.path.basename(src))
            shutil.copy2(src, dst)
            print("copied to: {}".format(dst))

    # Print beam info
    print("")
    print("--- Beam information ---")
    header = imhead(imagename + '.image', mode='list')
    try:
        bmaj = header['beammajor']
        bmin = header['beamminor']
        bpa = header['beampa']
        print("  Beam major: {:.3f} {}".format(bmaj['value'], bmaj['unit']))
        print("  Beam minor: {:.3f} {}".format(bmin['value'], bmin['unit']))
        print("  Beam PA:    {:.1f} {}".format(bpa['value'], bpa['unit']))
    except:
        print("  (per-plane beams; using a common beam)")
        try:
            pbeams = header['perplanebeams']
            median_beam = pbeams.get('median area beam', pbeams.get('*0', {}))
            print("  Median / first beam: {}".format(median_beam))
        except:
            print("  Could not read beam info; check manually with imhead")

    # Print basic image statistics
    print("")
    print("--- Image statistics ---")
    stats = imstat(imagename + '.image')
    print("  Max: {:.6f} Jy/beam".format(stats['max'][0]))
    print("  Min: {:.6f} Jy/beam".format(stats['min'][0]))
    print("  RMS: {:.6f} Jy/beam".format(stats['rms'][0]))

else:
    print("ERROR: image was not created!")

print("")
print("="*70)
print("  Step 2 finished")
print("="*70)
print("")
print("Next steps:")
print("  1. Inspect cube quality in CARTA: {}".format(imagename + '.fits'))
print("  2. Check that:")
print("     - the CO signal is clear (around channel 209-254)")
print("     - the H30α region is clean (around channel 43-79)")
print("     - line-free channels have uniform noise")
print("     - there are no black spots or stripe artifacts")
print("  3. The output is three FITS files:")
print("     - *_pbcor.fits:    pbcor cube (used for flux integration)")
print("     - *_nonpbcor.fits: non-pbcor cube (used to build the mask)")
print("     - *_pb.fits:       PB response")
print("  4. Once quality is confirmed, run step3_analyze.py")
